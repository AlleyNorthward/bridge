int filter_same(vector *vec, int *p) {
  int num = vec->data[*p];
  while (*p < vec->size && vec->data[*p] == num) {
    (*p)++;
  }

  return num;
}

void solve(vector *vec1, vector *vec2, vector *vec3) {
  int p1 = 0, p2 = 0, p3 = 0;
  while (p1 < vec1->size && p2 < vec2->size) {
    if (vec1->data[p1] < vec2->data[p2]) {

      vec3->data[p3++] = filter_same(vec1, &p1);
    } else if (vec1->data[p1] > vec2->data[p2]) {
      vec3->data[p3++] = filter_same(vec2, &p2);
    } else {

      vec3->data[p3++] = filter_same(vec1, &p1);
      filter_same(vec2, &p2);
    }
  }
  while (p1 < vec1->size) {
    vec3->data[p3++] = filter_same(vec1, &p1);
  }

  while (p2 < vec2->size) {
    vec3->data[p3++] = filter_same(vec2, &p2);
  }

  vec3->size = p3;
}
