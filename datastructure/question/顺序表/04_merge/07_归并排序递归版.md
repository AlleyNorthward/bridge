void merge(vector* vec, int left, int mid, int right){
  int* tmp = malloc(sizeof(int) * (right - left + 1));
  int p1 = left;
  int p2 = mid+1;
  int p3 = 0;

  while(p1 <= mid && p2 <= right){
    if(vec->data[p1] <= vec->data[p2]){
      tmp[p3++] = vec->data[p1++];
    } else {
      tmp[p3++] = vec->data[p2++];
    }
  }

  while(p1 <= mid){
    tmp[p3++] = vec->data[p1++];
  }

  while(p2 <= right){
    tmp[p3++] = vec->data[p2++];
  }

  for(int i = 0; i < p3; ++i){
    vec->data[left+i] = tmp[i];
  }
  free(tmp);
}

void mergeSort(vector* vec, int left, int right){
  if(left >= right)
    return;

  int mid = left + (right - left) / 2;
  mergeSort(vec, left, mid);
  mergeSort(vec, mid+1, right);
  merge(vec, left, mid, right);
}
