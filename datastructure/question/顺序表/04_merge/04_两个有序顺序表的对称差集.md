## question
- 给定两个升序顺序表 A 和 B.
- 存在于 A 中,但不存在于 B 中的元素;
- 存在于 B 中,但不存在于 A 中的元素;
- 结果不重复,保持升序.

## 答案
```c
void solve(vector *vec1, vector *vec2, vector *vec3) {
  int p1 = 0, p2 = 0, p3 = 0;
  while (p1 < vec1->size && p2 < vec2->size) {
    if (vec1->data[p1] < vec2->data[p2]) {
      int num = vec1->data[p1];
      vec3->data[p3++] = num;
      while (p1 < vec1->size && vec1->data[p1] == num) {
        ++p1;
      }
    } else if (vec1->data[p1] == vec2->data[p2]) {
      int num = vec1->data[p1];
      while (p1 < vec1->size && vec1->data[p1] == num)
        ++p1;

      while (p2 < vec2->size && vec2->data[p2] == num)
        ++p2;
    } else {
      int num = vec2->data[p2];
      vec3->data[p3++] = num;
      while (p2 < vec2->size && vec2->data[p2] == num) {
        ++p2;
      }
    }
  }

  while (p1 < vec1->size) {
    int num = vec1->data[p1];
    vec3->data[p3++] = num;
    while (p1 < vec1->size && vec1->data[p1] == num)
      ++p1;
  }

  while (p2 < vec2->size) {
    int num = vec2->data[p2];
    vec3->data[p3++] = num;
    while (p2 < vec2->size && vec2->data[p2] == num)
      ++p2;
  }

  vec3->size = p3;
}
```
